// ============================================================
// HELPERS
// ============================================================
function hex(color) { return color; } // already RGB objects

function frame(name, w, h) {
  const f = figma.createFrame();
  f.name = name;
  f.resize(w, h);
  f.fills = [{ type:"SOLID", color: C.bg }];
  f.clipsContent = true;
  return f;
}

function rect(parent, x, y, w, h, color, radius=0, opacity=1) {
  const r = figma.createRectangle();
  r.x = x; r.y = y; r.resize(w, h);
  r.fills = [{ type:"SOLID", color, opacity }];
  r.cornerRadius = radius;
  parent.appendChild(r);
  return r;
}

function text(parent, content, x, y, size, color, weight="Regular", w=null) {
  const t = figma.createText();
  t.x = x; t.y = y;
  t.characters = String(content);
  t.fontSize = size;
  t.fills = [{ type:"SOLID", color }];
  t.fontName = { family:"Readex Pro", style: weight };
  if (w) t.resize(w, t.height);
  parent.appendChild(t);
  return t;
}

function line(parent, x, y, w, color) {
  const l = figma.createLine();
  l.x = x; l.y = y;
  l.resize(w, 0);
  l.strokes = [{ type:"SOLID", color }];
  l.strokeWeight = 1;
  parent.appendChild(l);
  return l;
}

// SVG icon helper - appends an svg node
function icon(parent, svgStr, x, y, size=16) {
  const node = figma.createNodeFromSvg(svgStr);
  node.x = x; node.y = y;
  node.resize(size, size);
  parent.appendChild(node);
  return node;
}

// Status pill
function pill(parent, label, x, y, color) {
  const bg = figma.createFrame();
  bg.name = "pill-" + label;
  bg.x = x; bg.y = y;
  bg.resize(90, 22);
  bg.cornerRadius = 11;
  bg.fills = [{ type:"SOLID", color, opacity:0.14 }];
  bg.clipsContent = true;
  const t = figma.createText();
  t.characters = label;
  t.fontSize = 11;
  t.fontName = { family:"Readex Pro", style:"SemiBold" };
  t.fills = [{ type:"SOLID", color }];
  t.x = 8; t.y = 4;
  bg.appendChild(t);
  parent.appendChild(bg);
  return bg;
}

// Card / panel
function card(parent, x, y, w, h, radius=10, shadow=true) {
  const f = figma.createFrame();
  f.x = x; f.y = y;
  f.resize(w, h);
  f.cornerRadius = radius;
  f.fills = [{ type:"SOLID", color: C.card }];
  f.clipsContent = false;
  if (shadow) {
    f.effects = [{
      type: "DROP_SHADOW",
      color: { r:0, g:0, b:0, a:0.07 },
      offset: { x:0, y:2 }, radius:8, spread:0, visible:true, blendMode:"NORMAL"
    }];
  }
  parent.appendChild(f);
  return f;
}

// Top nav bar (truck-front inspired - wide horizontal bar)
function buildTopNav(parent, W, activeModule) {
  const navH = 56;
  const nav = figma.createFrame();
  nav.name = "TopNav";
  nav.x = 0; nav.y = 0;
  nav.resize(W, navH);
  nav.fills = [{ type:"SOLID", color: C.white }];
  nav.effects = [{
    type:"DROP_SHADOW",
    color:{r:0,g:0,b:0,a:0.06},
    offset:{x:0,y:2}, radius:6, spread:0, visible:true, blendMode:"NORMAL"
  }];
  parent.appendChild(nav);

  // Logo placeholder
  const logoBox = figma.createFrame();
  logoBox.x = 20; logoBox.y = 10;
  logoBox.resize(110, 36);
  logoBox.cornerRadius = 6;
  logoBox.fills = [{ type:"SOLID", color: C.red, opacity:0.08 }];
  nav.appendChild(logoBox);
  text(nav, "TRACKIFY", 30, 18, 14, C.red, "Bold");

  // Nav items
  const modules = [
    "Home",
    "Transactions",
    "Search",
    "Delivery",
    "Reports",
    "Security"
  ];
  const startX = 160;
  const itemW = 110;
  modules.forEach((mod, i) => {
    const isActive = mod === activeModule;
    const tx = startX + i * itemW;
    const t = text(nav, mod, tx, 18, 13, isActive ? C.red : C.textSub, isActive ? "SemiBold" : "Regular");
    if (isActive) {
      // Underline indicator
      const ind = figma.createRectangle();
      ind.x = tx; ind.y = navH - 3;
      ind.resize(t.width + 4, 3);
      ind.fills = [{ type:"SOLID", color: C.red }];
      ind.cornerRadius = 2;
      nav.appendChild(ind);
    }
  });

  // Right side: avatar circle
  const avCircle = figma.createEllipse();
  avCircle.x = W - 56; avCircle.y = 12;
  avCircle.resize(32, 32);
  avCircle.fills = [{ type:"SOLID", color: C.red, opacity:0.12 }];
  nav.appendChild(avCircle);
  text(nav, "R", W - 44, 19, 13, C.red, "Bold");

  return nav;
}

// Sidebar (left panel for sub-nav within module)
function buildSidebar(parent, y, H, items, activeItem) {
  const sb = card(parent, 0, y, 200, H - y, 0, false);
  sb.fills = [{ type:"SOLID", color: C.white }];
  sb.effects = [{
    type:"DROP_SHADOW",
    color:{r:0,g:0,b:0,a:0.05},
    offset:{x:2,y:0}, radius:8, spread:0, visible:true, blendMode:"NORMAL"
  }];

  let iy = 20;
  items.forEach(item => {
    const isActive = item === activeItem;
    if (isActive) {
      const bg = figma.createRectangle();
      bg.x = 8; bg.y = iy - 6;
      bg.resize(184, 32);
      bg.cornerRadius = 8;
      bg.fills = [{ type:"SOLID", color: C.red, opacity:0.10 }];
      sb.appendChild(bg);
    }
    text(sb, item, 20, iy, 13, isActive ? C.red : C.textSub, isActive ? "SemiBold" : "Regular");
    iy += 40;
  });
  return sb;
}

// Table header row
function tableHeader(parent, x, y, w, cols) {
  const rowH = 36;
  const bg = figma.createRectangle();
  bg.x = x; bg.y = y; bg.resize(w, rowH);
  bg.fills = [{ type:"SOLID", color: C.bg }];
  parent.appendChild(bg);

  let cx = x + 12;
  cols.forEach(col => {
    text(parent, col.label, cx, y + 10, 11, C.textSub, "SemiBold");
    cx += col.w;
  });
  return rowH;
}

// Table data row
function tableRow(parent, x, y, w, cols, data, alt=false) {
  const rowH = 40;
  const bg = figma.createRectangle();
  bg.x = x; bg.y = y; bg.resize(w, rowH);
  bg.fills = [{ type:"SOLID", color: alt ? C.cardAlt : C.card }];
  parent.appendChild(bg);

  line(parent, x, y + rowH - 1, w, C.border);

  let cx = x + 12;
  cols.forEach((col, ci) => {
    const val = data[ci] || "";
    // Status column special render
    if (col.label === "Status" && typeof val === "object" && val.label) {
      pill(parent, val.label, cx, y + 9, val.color);
    } else {
      text(parent, String(val), cx, y + 12, 12, C.text, "Regular");
    }
    cx += col.w;
  });
  return rowH;
}

// Floating modal overlay
function floatingModal(parent, x, y, w, h, title) {
  // Overlay
  const overlay = figma.createRectangle();
  overlay.x = 0; overlay.y = 56;
  overlay.resize(parent.width, parent.height - 56);
  overlay.fills = [{ type:"SOLID", color:{r:0,g:0,b:0}, opacity:0.25 }];
  parent.appendChild(overlay);

  const modal = card(parent, x, y, w, h, 14, true);
  modal.effects = [{
    type:"DROP_SHADOW",
    color:{r:0,g:0,b:0,a:0.18},
    offset:{x:0,y:8}, radius:32, spread:0, visible:true, blendMode:"NORMAL"
  }];

  // Header bar
  const header = figma.createRectangle();
  header.x = 0; header.y = 0; header.resize(w, 48);
  header.cornerRadius = 14;
  header.fills = [{ type:"SOLID", color: C.red }];
  modal.appendChild(header);

  text(modal, title, 20, 14, 14, C.white, "SemiBold");

  // Close X
  text(modal, "x", w - 32, 14, 14, C.white, "Regular");

  return modal;
}

// Stat card
function statCard(parent, x, y, w, h, label, value, sub, accentColor) {
  const c = card(parent, x, y, w, h, 12);
  // Left accent strip
  const strip = figma.createRectangle();
  strip.x = 0; strip.y = 0; strip.resize(4, h);
  strip.fills = [{ type:"SOLID", color: accentColor }];
  strip.cornerRadius = 12;
  c.appendChild(strip);

  text(c, label, 20, 16, 11, C.textSub, "Medium");
  text(c, value, 20, 32, 26, C.text, "Bold");
  if (sub) text(c, sub, 20, 62, 11, C.textSub, "Regular");
  return c;
}

// Button
function btn(parent, x, y, w, h, label, color, textColor) {
  const b = figma.createFrame();
  b.x = x; b.y = y; b.resize(w, h);
  b.cornerRadius = 8;
  b.fills = [{ type:"SOLID", color }];
  parent.appendChild(b);
  const t = text(b, label, 0, 0, 13, textColor, "SemiBold");
  t.x = Math.round((w - t.width) / 2);
  t.y = Math.round((h - t.height) / 2);
  return b;
}

// Input field
function inputField(parent, x, y, w, placeholder) {
  const f = figma.createFrame();
  f.x = x; f.y = y; f.resize(w, 36);
  f.cornerRadius = 8;
  f.fills = [{ type:"SOLID", color: C.card }];
  f.strokes = [{ type:"SOLID", color: C.border }];
  f.strokeWeight = 1;
  parent.appendChild(f);
  text(f, placeholder, 12, 10, 12, C.textHint, "Regular");
  return f;
}

// Section label
function sectionLabel(parent, x, y, label) {
  text(parent, label.toUpperCase(), x, y, 10, C.textHint, "SemiBold");
  return 20;
}
